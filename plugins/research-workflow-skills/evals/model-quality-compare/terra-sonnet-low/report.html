# Ledger design

`domain` validates event fields and converts decimal strings to integer cents;
formatting also stays there. Integer arithmetic keeps money exact without a
floating-point or decimal-context precision limit. `repository` owns the three
in-memory maps: current balances, per-tenant sequence numbers, and event records
(including the balance at application time). `service` orchestrates validation,
idempotency, sequencing, and the nonnegative-balance rule.

A batch operates on a shallow copy of all three maps. Their keys and values are
immutable, so no shared mutable entries can leak changes. Only after the input
iterator is exhausted successfully is the snapshot installed. Validation,
insufficient funds, or an iterator exception therefore leaves all original
maps unchanged, including idempotency records. Replays consult the snapshot, so
repeats within a batch work the same way as repeats across batches.

A future persistent or concurrent implementation would need a repository
transaction with isolation and an atomic commit instead of copying and replacing
maps; concurrent calls are not synchronized by this in-memory implementation.
The domain validation and the observable batch semantics should remain unchanged.

"""Validation and exact money conversion for ledger events."""
import re

_AMOUNT = re.compile(r'[+-]?[0-9]+(?:\.[0-9]{1,2})?\Z')


def amount(value):
    """Parse a plain decimal string into integer cents."""
    if not isinstance(value, str) or not _AMOUNT.fullmatch(value):
        raise ValueError('amount must be a plain decimal string with at most two decimal places')
    negative = value.startswith('-')
    unsigned = value.lstrip('+-')
    whole, _, fraction = unsigned.partition('.')
    cents = int(whole) * 100 + int((fraction + '00')[:2])
    return -cents if negative else cents


def format_amount(cents):
    sign = '-' if cents < 0 else ''
    cents = abs(cents)
    return f'{sign}{cents // 100}.{cents % 100:02d}'


def validate_event(event):
    if not isinstance(event, dict):
        raise ValueError('event must be a dictionary')
    tenant = event.get('tenant')
    event_id = event.get('event_id')
    seq = event.get('seq')
    if not isinstance(tenant, str) or not tenant.strip():
        raise ValueError('tenant must be a nonblank string')
    if not isinstance(event_id, str) or not event_id.strip():
        raise ValueError('event_id must be a nonblank string')
    if isinstance(seq, bool) or not isinstance(seq, int) or seq <= 0:
        raise ValueError('seq must be a positive integer')
    return tenant, event_id, seq, amount(event.get('amount'))

"""In-memory ledger state and transaction snapshots."""


class MemoryState:
    def __init__(self):
        self.balances = {}
        self.sequences = {}
        self.events = {}

    def copy(self):
        snapshot = MemoryState()
        snapshot.balances = self.balances.copy()
        snapshot.sequences = self.sequences.copy()
        snapshot.events = self.events.copy()
        return snapshot

ledger/service.py

from .domain import format_amount, validate_event
from .repository import MemoryState


class Ledger:
    def __init__(self):
        self.state = MemoryState()

    def balance(self, tenant):
        return format_amount(self.state.balances.get(tenant, 0))

    def apply_batch(self, events):
        # Work solely on a snapshot until the iterable has been exhausted.
        state = self.state.copy()
        result = []
        for event in events:
            tenant, event_id, seq, cents = validate_event(event)
            key = (tenant, event_id)
            if key in state.events:
                original_seq, original_cents, original_balance = state.events[key]
                if (seq, cents) != (original_seq, original_cents):
                    raise ValueError('event id reused with different contents')
                applied = False
                balance = original_balance
            else:
                if seq != state.sequences.get(tenant, 0) + 1:
                    raise ValueError('events must have consecutive sequence numbers')
                balance = state.balances.get(tenant, 0) + cents
                if balance < 0:
                    raise ValueError('insufficient funds')
                state.balances[tenant] = balance
                state.sequences[tenant] = seq
                state.events[key] = (seq, cents, balance)
                applied = True
            result.append({'tenant': tenant, 'event_id': event_id,
                           'applied': applied, 'balance': format_amount(balance)})
        self.state = state
        return result

class LedgerBehavior(unittest.TestCase):
    def test_interleaved_replay_and_exact_money(self):

        events = [event('a', 'x', 1, '9007199254740993.01'),

        self.assertEqual(ledger.apply_batch(iter(events)), [
            dict(tenant='a', event_id='x', applied=True, balance='9007199254740993.01'),
            dict(tenant='b', event_id='x', applied=True, balance='2.00'),
            dict(tenant='a', event_id='y', applied=True, balance='9007199254740993.00'),
            dict(tenant='a', event_id='x', applied=False, balance='9007199254740993.01')])
        self.assertEqual(ledger.balance('a'), '9007199254740993.00')
        self.assertEqual(ledger.apply_batch([event('b', 'x', 1, '2.00')])[0]['applied'], False)

    def test_atomic_failure_and_iterator_exception(self):
        ledger = Ledger()
        ledger.apply_batch([event('a', 'start', 1, '3')])
        def broken():
            yield event('a', 'next', 2, '1')
            yield event('b', 'first', 1, '5')
            raise RuntimeError('input failed')
        with self.assertRaisesRegex(RuntimeError, 'input failed'):
            ledger.apply_batch(broken())
        with self.assertRaises(ValueError):
            ledger.apply_batch([event('b', 'first', 1, '5'), event('a', 'next', 2, '-4')])
        self.assertEqual(ledger.balance('a'), '3.00')
        self.assertEqual(ledger.balance('b'), '0.00')
        self.assertTrue(ledger.apply_batch([event('a', 'next', 2, '1'), event('b', 'first', 1, '5')])[0]['applied'])

    def test_invalid_events_and_conflicts(self):
        ledger = Ledger()
        ledger.apply_batch([event('a', 'x', 1, '1')])
        bad = [event('a', 'x', 1, '2'), event('a', 'x', 2, '1'),
               event('a', 'y', 3, '1'), event(' ', 'y', 2, '1'),
               event('a', '', 2, '1'), event('a', 'y', True, '1'),
               event('a', 'y', 2, 1.0), event('a', 'y', 2, '1e2'),
               event('a', 'y', 2, 'NaN'), event('a', 'y', 2, '1.001'), {}]
        for item in bad:
            with self.subTest(item=item), self.assertRaises(ValueError):

        self.assertEqual(ledger.balance('a'), '1.00')
        self.assertEqual(ledger.apply_batch([]), [])
        self.assertEqual(ledger.apply_batch([event('a', 'y', 2, '-1')])[0]['balance'], '0.00')

    def test_input_unchanged(self):
        ledger = Ledger()
        item = event('a', 'x', 1, '+01.0')
        original = item.copy()
        ledger.apply_batch([item])
        self.assertEqual(item, original)
        self.assertFalse(ledger.apply_batch([event('a', 'x', 1, '1.00')])[0]['applied'])

# Education expenditure and income in the supplied World Bank data

## Questions and available evidence

The original questions ask in which regions increased education spending positively impacts GDP per capita, and what impact increased education expenditure has on GDP per capita in developing countries. The supplied data cannot identify those causal impacts or provide a comprehensive regional comparison. They contain two aggregate groups, Sub-Saharan Africa (SSF) and lower middle income (LMC), the latter an income classification rather than a geographic region or all developing countries. These are group-level observations, not a country-level panel; group membership may overlap.

The analysis uses the actual series codes and labels: X is adjusted savings: education expenditure (% of GNI), `NY.ADJ.AEDU.GN.ZS`; Y is GNI per capita (constant 2015 US dollars), `NY.GNP.PCAP.KD`. Education is a share of GNI, not spending per person or a share of total government expenditure. A change in that ratio need not represent a change in absolute education spending. GNI is not GDP. The metadata's general description instead claims government education expenditure as a percentage of total expenditure and GDP per capita in constant 2010 dollars; its detailed series descriptions and the CSV identify the different measures above. It also names the African code SSA, whereas the CSV uses SSF. Its description of education expenditure as money “saved for future education expenditures” should not be taken as establishing such a mechanism. Other indicators are present but are not used here.

## Method and results

Run `python3 analyze.py data.csv > results.json`. Within each group, pair finite observations of the two series for the same calendar year. For levels, calculate Pearson's r and the OLS slope with an intercept, treating education as X and income as Y. For differences, calculate those statistics on changes between consecutive paired calendar years only; missing years are not bridged. The slope is in constant 2015 US dollars of GNI per person per percentage-point difference in the education-to-GNI measure; the differenced slope relates annual changes in those quantities. These are descriptive contemporaneous associations, not adjusted effects. Results are rounded below; `results.json` retains computed precision.

| Group | Levels: years, n | Level r | Level slope | Consecutive annual changes: n | Change r | Change slope |
|---|---:|---:|---:|---:|---:|---:|
| Lower middle income (LMC) | 1975–2015, 41 | -0.481 | -635.07 | 40 | 0.079 | 11.06 |
| Sub-Saharan Africa (SSF) | 2011–2015, 5 | -0.966 | -159.13 | 4 | -0.456 | -80.48 |

Both level associations are negative. LMC annual changes show a very weak positive association; SSF changes show a negative association based on only four changes. Thus even the descriptive sign depends on the group and whether levels or changes are examined. Neither provides evidence that increased education spending causes a rise or fall in income, and these data cannot answer which regions experience a positive causal impact.

Differencing is a complementary choice to reduce the influence of persistent levels and common trends, not a causal design: it can amplify measurement noise and removes long-run information, while contemporaneous changes can miss delayed effects. Levels can reflect shared trends and serial dependence. Neither analysis controls for confounding, reverse causality, economic shocks, changing group composition, or other indicators, and no uncertainty estimates are supplied. The short SSF overlap is particularly fragile. The groups and aggregate series cannot reveal within-country relationships or identify all developing countries.

## Provenance

The supplied `metadata.json` attributes the data to World Development Indicators and identifies DiscoveryBench repository revision `c31fcf011e070f021a5f5b906896d0821f6880e8`, path `discoverybench/real/test/worldbank_education_gdp`, licensed ODC-By (see `SOURCE-LICENSE.txt`). analyze.py

#!/usr/bin/env python3
"""Summarize paired annual World Bank series for each group in a CSV."""
import csv
import json
import math
import re
import sys

INCOME = 'NY.GNP.PCAP.KD'
EDUCATION = 'NY.ADJ.AEDU.GN.ZS'


def number(value):
    try:
        result = float(value)
        return result if math.isfinite(result) else None
    except (TypeError, ValueError):
        return None


def statistics(pairs):
    n = len(pairs)
    result = {'n': n, 'pearson_r': None, 'ols_slope': None}
    if n < 2:
        return result
    xs, ys = zip(*pairs)
    mx, my = math.fsum(xs) / n, math.fsum(ys) / n
    dx = [x - mx for x in xs]
    dy = [y - my for y in ys]
    xx = math.fsum(x * x for x in dx)
    yy = math.fsum(y * y for y in dy)
    xy = math.fsum(x * y for x, y in zip(dx, dy))
    if xx:
        slope = xy / xx
        result['ols_slope'] = slope if math.isfinite(slope) else None
        if yy:
            r = (xy / math.sqrt(xx)) / math.sqrt(yy)
            result['pearson_r'] = r if math.isfinite(r) else None
    return result


def analyze(path):
    groups = {}
    with open(path, encoding='utf-8-sig', newline='') as source:
        reader = csv.DictReader(source)
        years = {}
        for name in reader.fieldnames:
            match = re.fullmatch(r'(\d{4})(?:\s*\[YR\d{4}\])?', name.strip())
            if match:
                years[int(match.group(1))] = name
        for row in reader:
            code, series = row['Country Code'].strip(), row['Series Code'].strip()
            if not code or series not in (INCOME, EDUCATION):
                continue
            group = groups.setdefault(code, {'label': row['Country Group'], 'series': {}})
            if series in group['series']:
                raise ValueError('Duplicate series for group: ' + code + ' ' + series)
            group['series'][series] = {year: number(row[column]) for year, column in years.items()}
    output = {}
    for code, group in sorted(groups.items()):
        if not all(s in group['series'] for s in (INCOME, EDUCATION)):
            continue
        x, y = group['series'][EDUCATION], group['series'][INCOME]
        paired = [(year, x[year], y[year]) for year in sorted(years)
                  if x[year] is not None and y[year] is not None]
        levels = [(a, b) for _, a, b in paired]
        differences = [(x1 - x0, y1 - y0)
                       for (t0, x0, y0), (t1, x1, y1) in zip(paired, paired[1:])
                       if t1 == t0 + 1]
        output[code] = {'label': group['label'], 'level': statistics(levels),
                        'difference': statistics(differences)}
    return {'series': {'income': INCOME, 'education': EDUCATION}, 'groups': output}


if __name__ == '__main__':
    if len(sys.argv) != 2:
        raise SystemExit('Usage: python3 analyze.py CSV_PATH')
    print(json.dumps(analyze(sys.argv[1]), indent=2, allow_nan=False))

# Ledger design

## Responsibilities
- `domain.py`: pure rules — event validation, exact decimal-string -> integer cents, formatting. No state.
- `repository.py`: owns state (balances, last seq per tenant, idempotency records keyed by `(tenant, event_id)` storing seq, cents and balance-after). `Transaction` stages writes in an overlay.
- `service.py` (`Ledger`): orchestration — parse, idempotency check, sequence check, funds check, stage, commit.

## Atomicity
`apply_batch` reads/writes only through a `Transaction`; later events in the batch see earlier staged ones. `commit()` runs only after the whole iterable is consumed and every event passed, so any ValueError or iterator exception leaves committed state untouched (commit is three dict updates with no validation that can fail).

## Money
Integer cents; no floats, so large values stay exact. Negative amounts are withdrawals; a balance below zero is rejected.

## Likely change: persistent storage
Only `repository.py` changes: `commit()` becomes a database transaction. Domain and service are unaffected. A new rule (e.g. overdraft limits) touches only the funds check in the service/domain.

"""Pure domain rules: event validation and money formatting. No state."""
import re

_AMOUNT = re.compile(r'-?(\d+)(?:\.(\d{1,2}))?\Z')


def parse_event(event):
    """Return (tenant, event_id, seq, cents) or raise ValueError."""
    if not isinstance(event, dict):
        raise ValueError('event must be a dict')
    tenant, event_id = event.get('tenant'), event.get('event_id')
    for name, v in (('tenant', tenant), ('event_id', event_id)):
        if not isinstance(v, str) or not v.strip():
            raise ValueError(f'invalid {name}')
    seq = event.get('seq')
    if type(seq) is not int or seq < 1:
        raise ValueError('invalid seq')
    return tenant, event_id, seq, cents(event.get('amount'))


def cents(value):
    if not isinstance(value, str) or not value.isascii():
        raise ValueError('invalid amount')
    m = _AMOUNT.match(value)
    if not m:
        raise ValueError('invalid amount')
    total = int(m.group(1)) * 100 + int((m.group(2) or '0').ljust(2, '0'))
    return -total if value.startswith('-') else total


def format_cents(n):
    sign = '-' if n < 0 else ''
    q, r = divmod(abs(n), 100)
    return f'{sign}{q}.{r:02d}'

ledger/repository.py

"""State ownership: committed state plus a staged transaction overlay."""


class MemoryState:
    def __init__(self):
        self.balances = {}   # tenant -> cents
        self.last_seq = {}   # tenant -> last applied seq
        self.applied = {}    # (tenant, event_id) -> (seq, cents, balance_after)

    def begin(self):
        return Transaction(self)


class Transaction:
    """Writes go to an overlay; nothing reaches MemoryState until commit()."""

    def __init__(self, state):
        self._s = state
        self._bal, self._seq, self._app = {}, {}, {}

    def balance(self, tenant):
        return self._bal.get(tenant, self._s.balances.get(tenant, 0))

    def last_seq(self, tenant):
        return self._seq.get(tenant, self._s.last_seq.get(tenant, 0))

    def record(self, key):
        return self._app.get(key) or self._s.applied.get(key)

    def put(self, tenant, event_id, seq, amount_cents, new_balance):
        self._bal[tenant] = new_balance
        self._seq[tenant] = seq
        self._app[(tenant, event_id)] = (seq, amount_cents, new_balance)

    def commit(self):
        self._s.balances.update(self._bal)
        self._s.last_seq.update(self._seq)
        self._s.applied.update(self._app)

"""Orchestration: validate, stage, commit atomically."""
from .domain import parse_event, format_cents
from .repository import MemoryState


class Ledger:
    def __init__(self):
        self.state = MemoryState()

    def balance(self, tenant):
        return format_cents(self.state.balances.get(tenant, 0))

    def apply_batch(self, events):
        tx = self.state.begin()
        result = []
        for event in events:  # iterator errors propagate before any commit
            tenant, event_id, seq, amt = parse_event(event)
            rec = tx.record((tenant, event_id))
            if rec is not None:
                if rec[:2] != (seq, amt):
                    raise ValueError('event key reused with different data')
                applied, bal = False, rec[2]
            else:
                if seq != tx.last_seq(tenant) + 1:
                    raise ValueError('non-consecutive seq')
                bal = tx.balance(tenant) + amt
                if bal < 0:
                    raise ValueError('insufficient funds')
                tx.put(tenant, event_id, seq, amt, bal)
                applied = True
            result.append({'tenant': tenant, 'event_id': event_id,
                           'applied': applied, 'balance': format_cents(bal)})
        tx.commit()
        return result

# Education spending and income: what these data support

## Questions
1. In what regions does increased education spending positively impact per capita GDP?
2. What is the impact of increased education expenditure on per capita GDP in developing countries?

## Data actually available
`data.csv` has two aggregate groups, not countries: Sub-Saharan Africa (`SSF`) and Lower middle income (`LMC`). Annual columns run 1975–2015 for six WDI series. I used:
- Income (Y): `NY.GNP.PCAP.KD`, GNI per capita, constant 2015 US$. This is GNI, not GDP.
- Education (X): `NY.ADJ.AEDU.GN.ZS`, adjusted savings: education expenditure, % of GNI.

The education series is nearly complete. Income is blank for SSF before 2011 (only 5 years, 2011–2015) but complete for LMC (41 years). Missing cells are blank here (no `..` in this file; the script also treats `..` as missing). Pairs use calendar years where both are finite. Differences use only consecutive years, with no bridging of gaps. OLS includes an intercept.

## Results (`results.json`, from `python3 analyze.py data.csv`)
| Group | Analysis | n | Pearson r | OLS slope (US$ per percentage point of GNI) |
|---|---|---|---|---|
| SSF | level | 5 | -0.966 | -159.1 |
| SSF | first difference | 4 | -0.456 | -80.5 |
| LMC | level | 41 | -0.481 | -635.1 |
| LMC | first difference | 40 | 0.079 | 11.1 |

## Analytical choice
Level correlations of two trending series are likely spurious: income trends upward while the education share drifts or falls, so level slopes mostly reflect time trends. The first-difference analysis removes slowly moving trends and is the more defensible one. With it, LMC shows essentially no association (r = 0.08, slope 11 with n = 40; no uncertainty is computed, but this is small and near zero). SSF's difference estimate rests on 4 pairs and is not informative.

## Conclusions warranted
- The data do **not** show that higher education spending is positively associated with income in either group. Level associations are negative; the LMC difference association is ~0.
- A region-by-region answer to question 1 is impossible beyond two aggregates; no positive association was found in either.
- Question 2 (impact) is a causal question. These data cannot answer it: only two aggregate series, no controls, no lags, no identification, reverse causality possible (education spending is a share of GNI, so GNI is in the denominator, which mechanically links the two), and education effects on income plausibly take years or decades.
- The "developing countries" are only two World Bank groups; findings are not country-level.

## Limitations and provenance
- SSF level/difference rest on 5 and 4 points; r = -0.97 at n = 5 is fragile and driven by 2011–2015 co-movement.
- No significance tests, lags, or other covariates (exports, enrollment, labor force were provided but not used).
- Metadata/data inconsistencies: metadata describes education spending as "% of total expenditure" and income as "per capita GDP (2010 US$)", while the data are % of GNI and GNI per capita in 2015 US$. Metadata names the Sub-Saharan code `SSA`; the file uses `SSF`. Metadata implies both groups span 1975–2015, but SSF income exists only for 2011–2015. I trusted the CSV series codes and names.
- Source: DiscoveryBench World Bank WDI extract supplied in `data.csv` and `metadata.json` (unchanged); see SOURCE-LICENSE.txt.

import csv, json, math, re, sys
INC, EDU = "NY.GNP.PCAP.KD", "NY.ADJ.AEDU.GN.ZS"

def num(s):
    try:
        v = float(str(s).strip())
    except ValueError:
        return None
    return v if math.isfinite(v) else None

def stats(pairs):
    n = len(pairs)
    out = {"n": n, "pearson_r": None, "ols_slope": None}
    if n < 2:
        return out
    mx = sum(p[0] for p in pairs) / n
    my = sum(p[1] for p in pairs) / n
    sxx = sum((p[0] - mx) ** 2 for p in pairs)
    syy = sum((p[1] - my) ** 2 for p in pairs)
    sxy = sum((p[0] - mx) * (p[1] - my) for p in pairs)
    xconst = len({p[0] for p in pairs}) == 1
    yconst = len({p[1] for p in pairs}) == 1
    if not xconst:
        out["ols_slope"] = sxy / sxx
    if not xconst and not yconst:
        out["pearson_r"] = max(-1.0, min(1.0, sxy / math.sqrt(sxx * syy)))
    return out

def main(path):
    with open(path, newline="", encoding="utf-8-sig") as f:
        rd = csv.reader(f)
        hdr = next(rd)
        ycols = []
        for i, h in enumerate(hdr):
            m = re.match(r"\s*(\d{4})\b", h)
            if m:
                ycols.append((int(m.group(1)), i))
        ix = {h.strip(): i for i, h in enumerate(hdr)}
        ig, ic, isr = ix["Country Group"], ix["Country Code"], ix["Series Code"]
        data, labels = {}, {}
        for r in rd:
            if len(r) <= max(ig, ic, isr):
                continue
            s = r[isr].strip()
            if s not in (INC, EDU):
                continue
            c = r[ic].strip()
            labels.setdefault(c, r[ig].strip())
            d = data.setdefault(c, {INC: {}, EDU: {}})[s]
            for y, i in ycols:
                v = num(r[i]) if i < len(r) else None
                if v is not None:
                    d[y] = v
    groups = {}
    for c in data:
        xs, ys = data[c][EDU], data[c][INC]
        yrs = sorted(set(xs) & set(ys))
        lev = [(xs[y], ys[y]) for y in yrs]
        dif = [(xs[b] - xs[a], ys[b] - ys[a]) for a, b in zip(yrs, yrs[1:]) if b == a + 1]
        groups[c] = {"label": labels[c], "level": stats(lev), "difference": stats(dif)}
    print(json.dumps({"series": {"income": INC, "education": EDU}, "groups": groups}))

if __name__ == "__main__":
    main(sys.argv[1])
